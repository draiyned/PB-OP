import { useState } from "react";
import { Plus, X, Shuffle, RotateCcw, Users, Check, Clock, Trophy } from "lucide-react";

const COURT_COLOR = "#2F6E5C"; // deep court green
const COURT_COLOR_IDLE = "#1E332E";
const COURT_COLOR_DONE = "#3D3626"; // muted clay-brown, distinct from active court green
const LINE_COLOR = "#F5EFD9"; // court line cream
const BALL_COLOR = "#E8D144"; // pickleball yellow
const CLAY = "#C4622D";

function pairKey(a, b) {
  return [a, b].sort().join("::");
}

function bestTeamSplit(four, partnerHistory) {
  const [p0, p1, p2, p3] = four;
  const options = [
    { teamA: [p0, p1], teamB: [p2, p3] },
    { teamA: [p0, p2], teamB: [p1, p3] },
    { teamA: [p0, p3], teamB: [p1, p2] },
  ];
  let best = options[0];
  let bestScore = Infinity;
  for (const opt of options) {
    const k1 = pairKey(opt.teamA[0].id, opt.teamA[1].id);
    const k2 = pairKey(opt.teamB[0].id, opt.teamB[1].id);
    const s = (partnerHistory[k1] || 0) + (partnerHistory[k2] || 0);
    if (s < bestScore) {
      bestScore = s;
      best = opt;
    }
  }
  return best;
}

export default function PickleballGenerator() {
  const [players, setPlayers] = useState([]);
  const [nameInput, setNameInput] = useState("");
  const [numCourts, setNumCourts] = useState(2);
  const [sessionStarted, setSessionStarted] = useState(false);
  const [courts, setCourts] = useState([]); // { teamA, teamB, matchNum } | null, per court index
  const [matchCounts, setMatchCounts] = useState([]); // per-court match number
  const [gamesPlayed, setGamesPlayed] = useState({}); // id -> count
  const [benchOrder, setBenchOrder] = useState({}); // id -> turn number they became free
  const [turnCounter, setTurnCounter] = useState(0);
  const [partnerHistory, setPartnerHistory] = useState({});
  const [history, setHistory] = useState([]); // completed matches, newest first
  const [error, setError] = useState("");
  const [confirmState, setConfirmState] = useState({}); // courtIndex -> "confirmFinish" | "recordScore" | "confirmNext"
  const [scoreDraft, setScoreDraft] = useState({}); // courtIndex -> { a: string, b: string }

  function addPlayer() {
    const name = nameInput.trim();
    if (!name) return;
    if (players.some((p) => p.name.toLowerCase() === name.toLowerCase())) {
      setError("That player's already on the list.");
      return;
    }
    const id = `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
    setPlayers([...players, { id, name }]);
    setGamesPlayed((g) => ({ ...g, [id]: 0 }));
    setBenchOrder((b) => ({ ...b, [id]: -1 })); // -1 = never played, highest priority
    setNameInput("");
    setError("");
  }

  function removePlayer(id) {
    setPlayers(players.filter((p) => p.id !== id));
  }

  function occupiedIds(courtsArr) {
    const s = new Set();
    courtsArr.forEach((c) => {
      if (!c) return;
      [...c.teamA, ...c.teamB].forEach((p) => s.add(p.id));
    });
    return s;
  }

  // pick the fairest available 4 (not already occupied on another court),
  // using local copies so we can fill multiple courts in one pass
  function pickFour(waitingPool, gp, bo) {
    const sorted = [...waitingPool].sort((a, b) => {
      const diff = (gp[a.id] || 0) - (gp[b.id] || 0);
      if (diff !== 0) return diff;
      return (bo[a.id] || 0) - (bo[b.id] || 0);
    });
    return sorted.slice(0, 4);
  }

  function startSession() {
    if (players.length < 4) {
      setError("Need at least 4 players.");
      return;
    }
    setError("");

    const gp = { ...gamesPlayed };
    const bo = { ...benchOrder };
    const ph = { ...partnerHistory };
    let counter = turnCounter;

    let waiting = [...players];
    const newCourts = [];
    const newMatchCounts = [];

    for (let i = 0; i < numCourts; i++) {
      if (waiting.length < 4) {
        newCourts.push(null);
        newMatchCounts.push(0);
        continue;
      }
      const four = pickFour(waiting, gp, bo);
      const { teamA, teamB } = bestTeamSplit(four, ph);
      four.forEach((p) => {
        gp[p.id] = (gp[p.id] || 0) + 1;
      });
      const kA = pairKey(teamA[0].id, teamA[1].id);
      const kB = pairKey(teamB[0].id, teamB[1].id);
      ph[kA] = (ph[kA] || 0) + 1;
      ph[kB] = (ph[kB] || 0) + 1;
      waiting = waiting.filter((p) => !four.find((f) => f.id === p.id));
      newCourts.push({ teamA, teamB });
      newMatchCounts.push(1);
    }

    setGamesPlayed(gp);
    setPartnerHistory(ph);
    setTurnCounter(counter);
    setCourts(newCourts);
    setMatchCounts(newMatchCounts);
    setHistory([]);
    setSessionStarted(true);
  }

  // step 1: user taps "Finished — next match" -> ask them to confirm
  function askFinishCourt(courtIndex) {
    setConfirmState((s) => ({ ...s, [courtIndex]: "confirmFinish" }));
  }

  function cancelFinishCourt(courtIndex) {
    setConfirmState((s) => ({ ...s, [courtIndex]: undefined }));
    setScoreDraft((s) => ({ ...s, [courtIndex]: undefined }));
    setError("");
  }

  // step 2: user confirms the match is over -> ask for the score before logging it
  function proceedToScore(courtIndex) {
    setScoreDraft((s) => ({ ...s, [courtIndex]: { a: "", b: "" } }));
    setConfirmState((s) => ({ ...s, [courtIndex]: "recordScore" }));
  }

  function updateScoreDraft(courtIndex, team, value) {
    setScoreDraft((s) => ({
      ...s,
      [courtIndex]: { ...s[courtIndex], [team]: value },
    }));
  }

  // step 3: user submits the score -> log it, free the players,
  // but don't fill the court yet — ask before generating the next match
  function saveScoreAndFinish(courtIndex) {
    const finished = courts[courtIndex];
    if (!finished) return;
    const draft = scoreDraft[courtIndex] || { a: "", b: "" };
    const scoreA = parseInt(draft.a, 10);
    const scoreB = parseInt(draft.b, 10);
    if (Number.isNaN(scoreA) || Number.isNaN(scoreB)) {
      setError("Enter a score for both teams.");
      return;
    }
    if (scoreA === scoreB) {
      setError("Scores can't be tied — pick a winner.");
      return;
    }

    let counter = turnCounter;
    const bo = { ...benchOrder };
    [...finished.teamA, ...finished.teamB].forEach((p) => {
      bo[p.id] = counter;
    });
    counter += 1;

    setHistory((h) => [
      {
        court: courtIndex,
        matchNum: matchCounts[courtIndex],
        teamA: finished.teamA,
        teamB: finished.teamB,
        scoreA,
        scoreB,
        winner: scoreA > scoreB ? "A" : "B",
      },
      ...h,
    ]);

    const newCourts = [...courts];
    newCourts[courtIndex] = null;
    setCourts(newCourts);
    setBenchOrder(bo);
    setTurnCounter(counter);
    setError("");
    setScoreDraft((s) => ({ ...s, [courtIndex]: undefined }));
    setConfirmState((s) => ({ ...s, [courtIndex]: "confirmNext" }));
  }

  // step 3: user taps "Generate next match" -> fill the court from whoever's free
  function generateNextForCourt(courtIndex) {
    const occupied = occupiedIds(courts);
    const waiting = players.filter((p) => !occupied.has(p.id));

    if (waiting.length < 4) {
      setError(
        `Court ${courtIndex + 1} is waiting on players — only ${waiting.length} free right now.`
      );
      return;
    }

    const gp = { ...gamesPlayed };
    const ph = { ...partnerHistory };

    const four = pickFour(waiting, gp, benchOrder);
    const { teamA, teamB } = bestTeamSplit(four, ph);
    four.forEach((p) => {
      gp[p.id] = (gp[p.id] || 0) + 1;
    });
    const kA = pairKey(teamA[0].id, teamA[1].id);
    const kB = pairKey(teamB[0].id, teamB[1].id);
    ph[kA] = (ph[kA] || 0) + 1;
    ph[kB] = (ph[kB] || 0) + 1;

    const newCourts = [...courts];
    newCourts[courtIndex] = { teamA, teamB };
    const newMatchCounts = [...matchCounts];
    newMatchCounts[courtIndex] = (newMatchCounts[courtIndex] || 0) + 1;

    setCourts(newCourts);
    setMatchCounts(newMatchCounts);
    setGamesPlayed(gp);
    setPartnerHistory(ph);
    setError("");
    setConfirmState((s) => ({ ...s, [courtIndex]: undefined }));
  }

  function resetAll() {
    setSessionStarted(false);
    setCourts([]);
    setMatchCounts([]);
    setHistory([]);
    setGamesPlayed(Object.fromEntries(players.map((p) => [p.id, 0])));
    setBenchOrder(Object.fromEntries(players.map((p) => [p.id, -1])));
    setTurnCounter(0);
    setPartnerHistory({});
    setError("");
    setConfirmState({});
    setScoreDraft({});
  }

  const occupied = occupiedIds(courts);
  const waitingPlayers = sessionStarted
    ? players.filter((p) => !occupied.has(p.id))
    : [];

  return (
    <div className="min-h-screen w-full" style={{ background: "#122320", color: LINE_COLOR }}>
      <div className="max-w-3xl mx-auto px-5 py-10">
        {/* Header */}
        <div className="flex items-center gap-4 mb-8">
          <svg width="56" height="56" viewBox="0 0 56 56" className="shrink-0">
            <rect x="2" y="2" width="52" height="52" rx="4" fill={COURT_COLOR} stroke={LINE_COLOR} strokeWidth="2" />
            <line x1="28" y1="2" x2="28" y2="54" stroke={LINE_COLOR} strokeWidth="2" />
            <line x1="14" y1="2" x2="14" y2="54" stroke={LINE_COLOR} strokeWidth="1.4" opacity="0.7" />
            <line x1="42" y1="2" x2="42" y2="54" stroke={LINE_COLOR} strokeWidth="1.4" opacity="0.7" />
            <circle cx="28" cy="28" r="4" fill={BALL_COLOR} />
          </svg>
          <div>
            <h1
              className="text-3xl tracking-tight"
              style={{ fontFamily: "Georgia, 'Times New Roman', serif", fontWeight: 700, color: LINE_COLOR }}
            >
              Open Play Rotator
            </h1>
            <p className="text-sm opacity-70 mt-0.5">
              Each court runs on its own clock — mark a court finished and it refills instantly.
            </p>
          </div>
        </div>

        {/* Setup */}
        <div className="rounded-lg p-5 mb-6" style={{ background: "#0D1C19", border: "1px solid rgba(245,239,217,0.12)" }}>
          <div className="flex items-center gap-2 mb-3">
            <Users size={16} style={{ color: BALL_COLOR }} />
            <span className="text-sm uppercase tracking-widest opacity-70">Players ({players.length})</span>
          </div>

          <div className="flex gap-2 mb-4">
            <input
              value={nameInput}
              onChange={(e) => setNameInput(e.target.value)}
              onKeyDown={(e) => e.key === "Enter" && addPlayer()}
              placeholder="Add a player name"
              className="flex-1 rounded px-3 py-2 text-sm outline-none"
              style={{ background: "#122320", border: "1px solid rgba(245,239,217,0.2)", color: LINE_COLOR }}
            />
            <button
              onClick={addPlayer}
              className="rounded px-3 py-2 flex items-center gap-1 text-sm font-medium transition-opacity hover:opacity-90"
              style={{ background: CLAY, color: LINE_COLOR }}
            >
              <Plus size={16} /> Add
            </button>
          </div>

          {players.length > 0 && (
            <div className="flex flex-wrap gap-2 mb-4">
              {players.map((p) => (
                <span
                  key={p.id}
                  className="flex items-center gap-1.5 rounded-full pl-3 pr-1.5 py-1 text-sm"
                  style={{ background: occupied.has(p.id) ? "rgba(232,209,68,0.15)" : "rgba(245,239,217,0.08)" }}
                >
                  {p.name}
                  <button
                    onClick={() => removePlayer(p.id)}
                    aria-label={`Remove ${p.name}`}
                    className="rounded-full p-0.5 hover:bg-white/10"
                  >
                    <X size={12} />
                  </button>
                </span>
              ))}
            </div>
          )}

          <label className="text-sm opacity-70 flex items-center gap-2 w-fit">
            Courts
            <input
              type="number"
              min={1}
              max={20}
              value={numCourts}
              disabled={sessionStarted}
              onChange={(e) => setNumCourts(Math.max(1, parseInt(e.target.value) || 1))}
              className="w-16 rounded px-2 py-1 text-sm outline-none disabled:opacity-50"
              style={{ background: "#122320", border: "1px solid rgba(245,239,217,0.2)", color: LINE_COLOR }}
            />
          </label>
        </div>

        {error && (
          <div className="rounded-md px-4 py-2 mb-4 text-sm" style={{ background: "rgba(196,98,45,0.18)", color: "#F0B896" }}>
            {error}
          </div>
        )}

        <div className="flex gap-3 mb-8">
          {!sessionStarted ? (
            <button
              onClick={startSession}
              className="flex items-center gap-2 rounded px-4 py-2.5 font-medium text-sm transition-opacity hover:opacity-90"
              style={{ background: BALL_COLOR, color: "#1B2A2A" }}
            >
              <Shuffle size={16} /> Start Session
            </button>
          ) : (
            <button
              onClick={resetAll}
              className="flex items-center gap-2 rounded px-4 py-2.5 text-sm opacity-70 hover:opacity-100"
              style={{ border: "1px solid rgba(245,239,217,0.2)" }}
            >
              <RotateCcw size={14} /> End session
            </button>
          )}
        </div>

        {sessionStarted && (
          <>
            {/* Courts */}
            <div className="grid gap-3 mb-8" style={{ gridTemplateColumns: "repeat(auto-fit, minmax(240px, 1fr))" }}>
              {courts.map((court, ci) => (
                <div
                  key={ci}
                  className="rounded-lg p-4 relative overflow-hidden flex flex-col justify-between"
                  style={{ background: court ? COURT_COLOR : COURT_COLOR_IDLE, minHeight: 128 }}
                >
                  {court && (
                    <div className="absolute inset-x-4 top-1/2 h-px" style={{ background: "rgba(245,239,217,0.35)" }} />
                  )}
                  <div className="flex items-center justify-between mb-2">
                    <span className="text-[10px] uppercase tracking-widest opacity-70">Court {ci + 1}</span>
                    {court && (
                      <span className="text-[10px] opacity-50">match {matchCounts[ci]}</span>
                    )}
                  </div>

                  {court ? (
                    <>
                      <div className="text-sm font-medium">{court.teamA.map((p) => p.name).join(" & ")}</div>
                      <div className="text-center text-[10px] opacity-60 my-1">vs</div>
                      <div className="text-sm font-medium">{court.teamB.map((p) => p.name).join(" & ")}</div>

                      {confirmState[ci] === "confirmFinish" ? (
                        <div className="mt-3">
                          <div className="text-xs opacity-80 mb-2">Mark this match finished?</div>
                          <div className="flex gap-2">
                            <button
                              onClick={() => proceedToScore(ci)}
                              className="flex items-center gap-1.5 rounded px-3 py-1.5 text-xs font-medium transition-opacity hover:opacity-90"
                              style={{ background: BALL_COLOR, color: "#1B2A2A" }}
                            >
                              <Check size={13} /> Yes, finished
                            </button>
                            <button
                              onClick={() => cancelFinishCourt(ci)}
                              className="rounded px-3 py-1.5 text-xs opacity-70 hover:opacity-100"
                              style={{ border: "1px solid rgba(245,239,217,0.3)" }}
                            >
                              Cancel
                            </button>
                          </div>
                        </div>
                      ) : confirmState[ci] === "recordScore" ? (
                        <div className="mt-3">
                          <div className="text-xs opacity-80 mb-2">What was the score?</div>
                          <div className="flex items-center gap-2 mb-2">
                            <label className="flex items-center gap-1.5 text-xs flex-1">
                              <span className="opacity-70 truncate">{court.teamA.map((p) => p.name).join(" & ")}</span>
                              <input
                                type="number"
                                min={0}
                                value={scoreDraft[ci]?.a ?? ""}
                                onChange={(e) => updateScoreDraft(ci, "a", e.target.value)}
                                className="w-14 rounded px-2 py-1 text-xs outline-none"
                                style={{ background: "#122320", border: "1px solid rgba(245,239,217,0.25)", color: LINE_COLOR }}
                              />
                            </label>
                          </div>
                          <div className="flex items-center gap-2 mb-3">
                            <label className="flex items-center gap-1.5 text-xs flex-1">
                              <span className="opacity-70 truncate">{court.teamB.map((p) => p.name).join(" & ")}</span>
                              <input
                                type="number"
                                min={0}
                                value={scoreDraft[ci]?.b ?? ""}
                                onChange={(e) => updateScoreDraft(ci, "b", e.target.value)}
                                className="w-14 rounded px-2 py-1 text-xs outline-none"
                                style={{ background: "#122320", border: "1px solid rgba(245,239,217,0.25)", color: LINE_COLOR }}
                              />
                            </label>
                          </div>
                          <div className="flex gap-2">
                            <button
                              onClick={() => saveScoreAndFinish(ci)}
                              className="flex items-center gap-1.5 rounded px-3 py-1.5 text-xs font-medium transition-opacity hover:opacity-90"
                              style={{ background: BALL_COLOR, color: "#1B2A2A" }}
                            >
                              <Check size={13} /> Save match
                            </button>
                            <button
                              onClick={() => cancelFinishCourt(ci)}
                              className="rounded px-3 py-1.5 text-xs opacity-70 hover:opacity-100"
                              style={{ border: "1px solid rgba(245,239,217,0.3)" }}
                            >
                              Cancel
                            </button>
                          </div>
                        </div>
                      ) : (
                        <button
                          onClick={() => askFinishCourt(ci)}
                          className="mt-3 self-start flex items-center gap-1.5 rounded px-3 py-1.5 text-xs font-medium transition-opacity hover:opacity-90"
                          style={{ background: BALL_COLOR, color: "#1B2A2A" }}
                        >
                          <Check size={13} /> Finished — next match
                        </button>
                      )}
                    </>
                  ) : confirmState[ci] === "confirmNext" ? (
                    <div>
                      <div className="text-xs opacity-70 mb-2">Match logged. Generate the next match?</div>
                      <button
                        onClick={() => generateNextForCourt(ci)}
                        className="flex items-center gap-1.5 rounded px-3 py-1.5 text-xs font-medium transition-opacity hover:opacity-90"
                        style={{ background: BALL_COLOR, color: "#1B2A2A" }}
                      >
                        <Shuffle size={13} /> Generate next match
                      </button>
                    </div>
                  ) : (
                    <div className="flex items-center gap-1.5 text-xs opacity-50">
                      <Clock size={13} /> Waiting for players
                    </div>
                  )}
                </div>
              ))}
            </div>

            {/* Waiting / bench */}
            <div className="mb-8">
              <div className="text-xs uppercase tracking-widest opacity-50 mb-2">
                Waiting ({waitingPlayers.length})
              </div>
              <div className="flex flex-wrap gap-2">
                {waitingPlayers.length === 0 ? (
                  <span className="text-sm opacity-40">Everyone's on a court.</span>
                ) : (
                  waitingPlayers.map((p) => (
                    <span key={p.id} className="rounded-full px-3 py-1 text-sm" style={{ background: "rgba(245,239,217,0.08)" }}>
                      {p.name}
                    </span>
                  ))
                )}
              </div>
            </div>

            {/* Match history */}
            {history.length > 0 && (
              <div>
                <div className="text-xs uppercase tracking-widest opacity-50 mb-2">Completed matches</div>
                <div className="grid gap-3" style={{ gridTemplateColumns: "repeat(auto-fit, minmax(240px, 1fr))" }}>
                  {history.map((h, i) => (
                    <div
                      key={i}
                      className="rounded-lg p-4 relative overflow-hidden"
                      style={{ background: COURT_COLOR_DONE }}
                    >
                      <div className="absolute inset-x-4 top-1/2 h-px" style={{ background: "rgba(245,239,217,0.2)" }} />
                      <div className="flex items-center justify-between mb-2">
                        <span className="text-[10px] uppercase tracking-widest opacity-60">Court {h.court + 1}</span>
                        <span className="text-[10px] opacity-40 flex items-center gap-1">
                          <Check size={11} /> match {h.matchNum}
                        </span>
                      </div>
                      <div
                        className="text-sm font-medium opacity-80 flex items-center justify-between"
                        style={{ opacity: h.winner === "A" ? 1 : 0.6 }}
                      >
                        <span className="flex items-center gap-1">
                          {h.winner === "A" && <Trophy size={12} style={{ color: BALL_COLOR }} />}
                          {h.teamA.map((p) => p.name).join(" & ")}
                        </span>
                        {h.scoreA !== undefined && (
                          <span className="text-xs font-semibold ml-2">{h.scoreA}</span>
                        )}
                      </div>
                      <div className="text-center text-[10px] opacity-40 my-1">vs</div>
                      <div
                        className="text-sm font-medium opacity-80 flex items-center justify-between"
                        style={{ opacity: h.winner === "B" ? 1 : 0.6 }}
                      >
                        <span className="flex items-center gap-1">
                          {h.winner === "B" && <Trophy size={12} style={{ color: BALL_COLOR }} />}
                          {h.teamB.map((p) => p.name).join(" & ")}
                        </span>
                        {h.scoreB !== undefined && (
                          <span className="text-xs font-semibold ml-2">{h.scoreB}</span>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
}
